## 3. packages/api-client — TypeScript Client

### Client Factory

```typescript
// packages/api-client/src/client.ts
import createClient from 'openapi-fetch'
import type { paths } from './schema'  // generated — do not edit

function resolveTenantFromHostname(hostname: string): string {
  // e.g. acme.rcm.app → "acme", localhost → "dev"
  const parts = hostname.split('.')
  return parts.length >= 3 ? parts[0] : 'dev'
}

export const apiClient = createClient<paths>({
  baseUrl: '/api/v1',
  credentials: 'include',  // sends httpOnly cookie on every request
})

// Middleware: inject X-Tenant-ID on every outbound request
apiClient.use({
  onRequest({ request }) {
    const tenant = resolveTenantFromHostname(window.location.hostname)
    request.headers.set('X-Tenant-ID', tenant)
    return request
  },
  onResponse({ response }) {
    // Bubble RFC 7807 errors as thrown objects for TanStack Query to catch
    if (!response.ok) {
      return response.json().then((body) => {
        const err = new Error(body?.detail ?? response.statusText)
        Object.assign(err, { status: response.status, body })
        throw err
      })
    }
    return response
  },
})

export type { paths }
```

## 5. AuthProvider + RouteGuard

```tsx
// packages/ui/src/auth/AuthProvider.tsx
import { createContext, useContext, useEffect, useState, ReactNode } from 'react'
import { apiClient } from '@rcm/api-client'
import type { components } from '@rcm/api-client'
import { UserRole } from '@rcm/shared-types'
import { FullPageSpinner } from '../components/Spinner'
import { Navigate } from 'react-router-dom'  // or next/navigation in web-booking

type UserProfile = components['schemas']['UserProfile']

interface AuthContextValue {
  user: UserProfile | null
  isLoading: boolean
  setUser: (u: UserProfile | null) => void
  logout: () => Promise<void>
}

const AuthContext = createContext<AuthContextValue | null>(null)

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<UserProfile | null>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    // httpOnly cookie is sent automatically — never reads localStorage
    apiClient
      .GET('/auth/me')
      .then(({ data }) => setUser(data ?? null))
      .catch(() => setUser(null))
      .finally(() => setIsLoading(false))
  }, [])

  async function logout() {
    await apiClient.POST('/auth/logout')
    setUser(null)
  }

  return (
    <AuthContext.Provider value={{ user, isLoading, setUser, logout }}>
      {children}
    </AuthContext.Provider>
  )
}

export function useAuth(): AuthContextValue {
  const ctx = useContext(AuthContext)
  if (!ctx) throw new Error('useAuth must be used inside AuthProvider')
  return ctx
}

// RouteGuard — wrap any protected subtree
export function RouteGuard({
  roles = [],
  children,
  redirectTo = '/login',
}: {
  roles?: UserRole[]
  children: ReactNode
  redirectTo?: string
}) {
  const { user, isLoading } = useAuth()

  if (isLoading) return <FullPageSpinner />
  if (!user) return <Navigate to={redirectTo} replace />
  if (roles.length > 0 && !roles.some((r) => user.roles.includes(r))) {
    return <Navigate to="/unauthorized" replace />
  }
  return <>{children}</>
}
```

### Typed Wrappers

```typescript
// packages/ui/src/query/hooks.ts
import { useQuery, useMutation, UseQueryOptions, UseMutationOptions } from '@tanstack/react-query'
import { apiClient } from '@rcm/api-client'
import type { paths } from '@rcm/api-client'

// Generic typed query wrapper — path and method inferred from paths
type GetPaths = {
  [P in keyof paths]: paths[P] extends { get: unknown } ? P : never
}[keyof paths]

export function useApiQuery<
  TPath extends GetPaths,
  TData = paths[TPath]['get'] extends { responses: { 200: { content: { 'application/json': infer D } } } } ? D : unknown,
>(
  path: TPath,
  params?: Record<string, unknown>,
  options?: Omit<UseQueryOptions<TData>, 'queryKey' | 'queryFn'>
) {
  return useQuery<TData>({
    queryKey: [path, params],
    queryFn: () =>
      apiClient
        .GET(path as string, { params: params as never })
        .then(({ data }) => data as TData),
    ...options,
  })
}

// Generic typed mutation wrapper
export function useApiMutation<TBody, TData = unknown>(
  method: 'POST' | 'PUT' | 'PATCH' | 'DELETE',
  path: string,
  options?: UseMutationOptions<TData, Error, TBody>
) {
  return useMutation<TData, Error, TBody>({
    mutationFn: (body) =>
      (apiClient as never)[method](path, { body }).then(
        ({ data }: { data: TData }) => data
      ),
    ...options,
  })
}
```

### Counter Store

```typescript
// apps/web-counter/src/stores/counterStore.ts
import { create } from 'zustand'
import { persist, createJSONStorage } from 'zustand/middleware'

export type ModalState = 'closed' | 'checkout' | 'check-in' | 'damage' | 'swap'

export interface OfflineQueueItem {
  id: string
  endpoint: '/checkout' | '/check-in'
  payload: Record<string, unknown>
  queuedAt: string  // ISO 8601
  retryCount: number
}

interface CounterState {
  // Modal
  modalState: ModalState
  setModalState: (s: ModalState) => void

  // Multi-step wizard position
  currentStep: number
  totalSteps: number
  setStep: (n: number) => void
  nextStep: () => void
  prevStep: () => void
  resetStepper: () => void

  // Offline queue (persisted to IndexedDB via localStorage fallback)
  offlineQueue: OfflineQueueItem[]
  enqueueOfflineItem: (item: Omit<OfflineQueueItem, 'id' | 'queuedAt' | 'retryCount'>) => void
  removeOfflineItem: (id: string) => void
  clearOfflineQueue: () => void
}

export const useCounterStore = create<CounterState>()(
  persist(
    (set, get) => ({
      modalState: 'closed',
      setModalState: (s) => set({ modalState: s }),

      currentStep: 0,
      totalSteps: 6,
      setStep: (n) => set({ currentStep: n }),
      nextStep: () => set((s) => ({ currentStep: Math.min(s.currentStep + 1, s.totalSteps - 1) })),
      prevStep: () => set((s) => ({ currentStep: Math.max(s.currentStep - 1, 0) })),
      resetStepper: () => set({ currentStep: 0 }),

      offlineQueue: [],
      enqueueOfflineItem: (item) =>
        set((s) => ({
          offlineQueue: [
            ...s.offlineQueue,
            {
              ...item,
              id: crypto.randomUUID(),
              queuedAt: new Date().toISOString(),
              retryCount: 0,
            },
          ],
        })),
      removeOfflineItem: (id) =>
        set((s) => ({ offlineQueue: s.offlineQueue.filter((i) => i.id !== id) })),
      clearOfflineQueue: () => set({ offlineQueue: [] }),
    }),
    {
      name: 'rcm-counter-store',
      storage: createJSONStorage(() => localStorage),
      partialize: (s) => ({ offlineQueue: s.offlineQueue }), // only persist queue
    }
  )
)
```

```typescript
// apps/web-admin/src/stores/adminStore.ts
import { create } from 'zustand'

export type AdminSection = 'fleet' | 'reservations' | 'customers' | 'pricing' | 'reports' | 'settings'

interface ActiveFilters {
  status?: string[]
  locationId?: string
  dateRange?: { from: string; to: string }
  search?: string
}

interface AdminState {
  // Sidebar
  sidebarCollapsed: boolean
  toggleSidebar: () => void
  setSidebarCollapsed: (v: boolean) => void

  // Active filters per section (reset on section change)
  activeSection: AdminSection
  activeFilters: ActiveFilters
  setActiveSection: (s: AdminSection) => void
  setActiveFilters: (f: Partial<ActiveFilters>) => void
  resetFilters: () => void

  // Bulk selection (e.g. bulk status change)
  selectedItems: string[]
  toggleItemSelection: (id: string) => void
  selectAll: (ids: string[]) => void
  clearSelection: () => void
}

export const useAdminStore = create<AdminState>()((set) => ({
  sidebarCollapsed: false,
  toggleSidebar: () => set((s) => ({ sidebarCollapsed: !s.sidebarCollapsed })),
  setSidebarCollapsed: (v) => set({ sidebarCollapsed: v }),

  activeSection: 'fleet',
  activeFilters: {},
  setActiveSection: (s) => set({ activeSection: s, activeFilters: {}, selectedItems: [] }),
  setActiveFilters: (f) => set((s) => ({ activeFilters: { ...s.activeFilters, ...f } })),
  resetFilters: () => set({ activeFilters: {} }),

  selectedItems: [],
  toggleItemSelection: (id) =>
    set((s) => ({
      selectedItems: s.selectedItems.includes(id)
        ? s.selectedItems.filter((i) => i !== id)
        : [...s.selectedItems, id],
    })),
  selectAll: (ids) => set({ selectedItems: ids }),
  clearSelection: () => set({ selectedItems: [] }),
}))
```

### Service Worker (Workbox)

```typescript
// apps/web-counter/src/service-worker.ts
import { precacheAndRoute } from 'workbox-precaching'
import { registerRoute } from 'workbox-routing'
import { NetworkFirst, CacheFirst } from 'workbox-strategies'
import { BackgroundSyncPlugin } from 'workbox-background-sync'

// Precache all static assets (injected by vite-plugin-pwa)
precacheAndRoute(self.__WB_MANIFEST)

// Background sync for write operations — retries for up to 24 hours
const bgSyncPlugin = new BackgroundSyncPlugin('offline-write-queue', {
  maxRetentionTime: 24 * 60,  // minutes (= 24 hours)
})

// POST /checkout — queue when offline, sync when reconnected
registerRoute(
  ({ request, url }) =>
    request.method === 'POST' && url.pathname.includes('/checkout'),
  new NetworkFirst({ plugins: [bgSyncPlugin] })
)

// POST /check-in — same offline queue
registerRoute(
  ({ request, url }) =>
    request.method === 'POST' && url.pathname.includes('/check-in'),
  new NetworkFirst({ plugins: [bgSyncPlugin] })
)

// API GETs — NetworkFirst with 10s timeout, falls back to cache
registerRoute(
  ({ url }) => url.pathname.startsWith('/api/'),
  new NetworkFirst({
    cacheName: 'api-cache',
    networkTimeoutSeconds: 10,
  })
)

// Static assets (JS, CSS, fonts) — CacheFirst
registerRoute(
  ({ request }) =>
    request.destination === 'script' ||
    request.destination === 'style' ||
    request.destination === 'font',
  new CacheFirst({
    cacheName: 'static-assets',
    plugins: [],
  })
)
```

### Availability Grid (WebSocket + TanStack Query)

```tsx
// apps/web-counter/src/features/availability/AvailabilityGrid.tsx
import { useEffect } from 'react'
import { useQuery } from '@tanstack/react-query'
import { queryClient } from '@rcm/ui/query'
import { apiClient } from '@rcm/api-client'

function applyAvailabilityPatch(
  current: AvailabilityData | undefined,
  event: AvailabilityEvent
): AvailabilityData {
  if (!current) return current as never
  return {
    ...current,
    classes: current.classes.map((cls) =>
      cls.classId === event.classId
        ? { ...cls, availableCount: event.availableCount }
        : cls
    ),
  }
}

export function AvailabilityGrid({ locationId }: { locationId: string }) {
  const { data, isLoading } = useQuery({
    queryKey: ['availability', locationId],
    queryFn: () =>
      apiClient
        .GET('/fleet/availability/{locationId}', { params: { path: { locationId } } })
        .then((r) => r.data),
    refetchInterval: 30_000,  // poll every 30s as WebSocket fallback
    staleTime: 25_000,
  })

  // WebSocket subscription for real-time patch updates
  useEffect(() => {
    const ws = new WebSocket(
      `${location.protocol === 'https:' ? 'wss' : 'ws'}://${location.host}/api/v1/ws/fleet/${locationId}`
    )
    ws.onmessage = (e) => {
      const event: AvailabilityEvent = JSON.parse(e.data)
      queryClient.setQueryData(
        ['availability', locationId],
        (old: AvailabilityData) => applyAvailabilityPatch(old, event)
      )
    }
    ws.onerror = () => ws.close()
    // Reconnect handled by browser on close — TQ poll is the fallback
    return () => ws.close()
  }, [locationId])

  if (isLoading) return <Skeleton variant="table" rows={12} />

  // CSS Grid: class rows (auto) × time-slot columns (repeat(var(--slots), 1fr))
  return (
    <div
      className="grid gap-px bg-border"
      style={{ gridTemplateColumns: `auto repeat(${SLOT_COUNT}, minmax(60px, 1fr))` }}
    >
      {data?.classes.map((cls) => (
        <AvailabilityRow key={cls.classId} row={cls} />
      ))}
    </div>
  )
}
```

### OFFLINE Banner

```tsx
// apps/web-counter/src/components/OfflineBanner.tsx
import { useEffect, useState } from 'react'

export function OfflineBanner() {
  const [offline, setOffline] = useState(!navigator.onLine)

  useEffect(() => {
    const on = () => setOffline(false)
    const off = () => setOffline(true)
    window.addEventListener('online', on)
    window.addEventListener('offline', off)

    // Also listen for service worker message (more reliable in some browsers)
    navigator.serviceWorker?.addEventListener('message', (e) => {
      if (e.data?.type === 'NETWORK_STATUS') setOffline(!e.data.online)
    })

    return () => {
      window.removeEventListener('online', on)
      window.removeEventListener('offline', off)
    }
  }, [])

  if (!offline) return null

  return (
    <div
      role="alert"
      aria-live="assertive"
      className="fixed top-0 inset-x-0 z-50 flex items-center justify-center gap-2 bg-amber-500 px-4 py-2 text-sm font-semibold text-amber-950"
    >
      <span aria-hidden>⚠</span>
      OFFLINE — Checkout and check-in will sync automatically when reconnected
    </div>
  )
}
```

## 11. WCAG 2.1 AA Checklist

- **Colour contrast 4.5:1 (text) / 3:1 (UI components)** — Tailwind colour tokens enforced via `tailwind-plugin-a11y`; custom `cn()` utility validates at build time. All shadcn/ui defaults audited and overridden where needed.
- **Keyboard navigation** — All interactive elements reachable via Tab. Custom components (Stepper, DataTable, DatePicker) implement roving tabindex via `onKeyDown` arrow-key handlers. No keyboard traps except intentional modal focus traps.
- **Focus trap in modals** — `@radix-ui/react-dialog` handles focus trap automatically. Custom modals use `focus-trap-react`.
- **Focus visible indicator** — `:focus-visible` ring on all interactive elements. Never `outline: none` without a visible replacement.
- **Skip-to-content link** — All apps render `<a href="#main-content" className="sr-only focus:not-sr-only">Skip to content</a>` as first element in `<body>`.
- **Semantic landmarks** — `<header>`, `<nav>`, `<main id="main-content">`, `<aside>`, `<footer>` used correctly. No `<div>` for interactive roles.
- **ARIA roles on custom widgets** — Stepper: `role="tablist"` / `role="tab"` / `aria-selected` / `aria-controls`. DataTable: `role="grid"` with `aria-sort`. AvailabilityGrid: `role="grid"` with `aria-label`.
- **Live regions for async feedback** — Form errors: `role="alert" aria-live="polite" aria-atomic="true"`. Toast notifications: `role="status"`. Offline banner: `role="alert" aria-live="assertive"`.
- **Form labels** — Every `<input>` has either a visible `<label>` (preferred) or `aria-label`. `aria-describedby` links `FormDescription` and `FormError` to inputs.
- **Images and icons** — All decorative SVGs: `aria-hidden="true"`. Icon-only buttons: `aria-label` required (enforced by ESLint rule `jsx-a11y/interactive-supports-focus`).
- **Touch targets 44×44px minimum** — Buttons and interactive elements have `min-h-[44px] min-w-[44px]` utility classes. Enforced in `Button` and `Checkbox` components by default.
- **Accessible error states** — Validation errors associated with inputs via `aria-invalid="true"` and `aria-errormessage={errorId}`. Never colour-only error indication.
- **Accessible tables** — `<th scope="col">` / `<th scope="row">` on all table headers. `<caption>` for data tables. Sortable columns announce sort direction via `aria-sort`.
- **Motion / animation** — All animations respect `prefers-reduced-motion`. Tailwind `motion-safe:` and `motion-reduce:` variants used throughout.
- **Language attribute** — `<html lang="en">` set in every app's root layout.
- **axe-core CI gate** — Zero violations required; blocks merge (see §12).
